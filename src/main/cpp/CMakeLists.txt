cmake_minimum_required(VERSION 3.22)
project(infiniplay LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
get_filename_component(PLAY_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
include("${PLAY_ROOT}/build/native-source/quicknes-sources.cmake")
include("${PLAY_ROOT}/build/native-source/gambatte-sources.cmake")
add_library(quicknes SHARED ${QUICKNES_SOURCES})
target_include_directories(quicknes PRIVATE "${PLAY_ROOT}/third_party/quicknes" "${PLAY_ROOT}/third_party/quicknes/nes_emu" "${PLAY_ROOT}/third_party/quicknes/libretro" "${PLAY_ROOT}/third_party/quicknes/libretro/libretro-common/include")
target_compile_definitions(quicknes PRIVATE __LIBRETRO__ NDEBUG)
add_library(gambatte SHARED ${GAMBATTE_SOURCES})
set(GB "${PLAY_ROOT}/third_party/gambatte")
target_include_directories(gambatte PRIVATE "${GB}/libgambatte/src" "${GB}/libgambatte/include" "${GB}/common" "${GB}/common/resample" "${GB}/libgambatte/libretro" "${GB}/libgambatte/libretro-common/include")
target_compile_definitions(gambatte PRIVATE __LIBRETRO__ HAVE_STDINT_H HAVE_INTTYPES_H CC_RESAMPLER_NO_HIGHPASS VIDEO_RGB565 HAVE_STRL VFS_FRONTEND NDEBUG)
include("${PLAY_ROOT}/build/native-source/fceumm-sources.cmake")
list(FILTER FCEUMM_SOURCES EXCLUDE REGEX "/src/fds\\.c$|/drivers/libretro/libretro\\.c$")
add_library(fceumm SHARED ${FCEUMM_SOURCES} fds_adapter.c fceumm_adapter.c)
set(FC "${PLAY_ROOT}/third_party/fceumm/src")
target_include_directories(fceumm PRIVATE "${FC}" "${FC}/input" "${FC}/boards" "${FC}/drivers/libretro" "${FC}/drivers/libretro/libretro-common/include")
target_compile_definitions(fceumm PRIVATE __LIBRETRO__ FCEU_VERSION_NUMERIC=9900 FRONTEND_SUPPORTS_RGB565 HAVE_STRL VFS_FRONTEND)
target_link_libraries(fceumm PRIVATE m)
include("${PLAY_ROOT}/build/native-source/extra-cores.cmake")
get_target_property(PCE_SOURCES pce SOURCES)
list(FILTER PCE_SOURCES EXCLUDE REGEX "/third_party/pce/libretro\\.c$")
set_property(TARGET pce PROPERTY SOURCES ${PCE_SOURCES} "${CMAKE_CURRENT_SOURCE_DIR}/pce_adapter.c")
target_link_libraries(atari800 PRIVATE log)
target_compile_definitions(picodrive PRIVATE Z7_ST ZSTD_DISABLE_ASM USE_TREMOR)
target_include_directories(picodrive PRIVATE "${PLAY_ROOT}/third_party/picodrive/platform/common/tremor" "${PLAY_ROOT}/third_party/picodrive/pico/cd/libchdr/deps/zstd-1.5.6/lib")
target_compile_options(picodrive PRIVATE -fno-strict-aliasing)
set(HOST "${PLAY_ROOT}/third_party/libretrodroid/libretrodroid/src/main/cpp")
add_subdirectory("${HOST}" "host")
get_target_property(HOST_SOURCES libretrodroid SOURCES)
list(FILTER HOST_SOURCES EXCLUDE REGEX "^(video|input)\\.cpp$")
set_property(TARGET libretrodroid PROPERTY SOURCES ${HOST_SOURCES} "${PLAY_ROOT}/build/native-source/video.cpp" "${PLAY_ROOT}/build/native-source/input.cpp")
target_compile_options(libretrodroid PRIVATE "$<$<COMPILE_LANGUAGE:CXX>:-include>" "$<$<COMPILE_LANGUAGE:CXX>:functional>")
target_include_directories(libretrodroid PRIVATE "${HOST}" "${CMAKE_CURRENT_SOURCE_DIR}")
foreach(core quicknes gambatte fceumm libretrodroid genesis picodrive pce handy stella atari800 prosystem snes9x vb geolith)
    target_compile_options(${core} PRIVATE -O2)
    target_link_options(${core} PRIVATE "-Wl,-z,max-page-size=16384" "-Wl,--no-undefined")
endforeach()
if(PLAY_TERMUX)
    target_include_directories(libretrodroid PRIVATE "${PLAY_JNI_INCLUDE}")
    target_link_directories(libretrodroid PRIVATE /system/lib64)
    foreach(core quicknes gambatte fceumm libretrodroid genesis picodrive pce handy stella atari800 prosystem snes9x vb geolith)
        target_link_directories(${core} PRIVATE /system/lib64 "${PLAY_JNI_INCLUDE}/../lib")
    endforeach()
endif()
