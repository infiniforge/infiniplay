# Use mGBA's supported CMake target without desktop frontends or host-only dependencies.
function(play_add_mgba)
    set(LIBMGBA_ONLY OFF)
    set(BUILD_LIBRETRO ON CACHE BOOL "" FORCE)
    set(SKIP_LIBRARY ON CACHE BOOL "" FORCE)
    set(DISABLE_DEPS ON CACHE BOOL "" FORCE)
    foreach(option BUILD_QT BUILD_SDL BUILD_SHARED BUILD_STATIC BUILD_LTO BUILD_GL BUILD_GLES2 BUILD_GLES3
            BUILD_TEST BUILD_SUITE BUILD_CINEMA BUILD_PERF BUILD_HEADLESS BUILD_EXAMPLE BUILD_PYTHON
            ENABLE_DEBUGGERS ENABLE_GDB_STUB ENABLE_SCRIPTING USE_LUA USE_FFMPEG USE_LIBZIP USE_MINIZIP
            USE_PNG USE_ZLIB USE_LZMA USE_SQLITE3 USE_ELF USE_DISCORD_RPC USE_JSON_C USE_FREETYPE USE_EPOXY)
        set(${option} OFF CACHE BOOL "" FORCE)
    endforeach()
    add_subdirectory("${PLAY_ROOT}/third_party/mgba" "mgba")
    set_target_properties(mgba_libretro PROPERTIES OUTPUT_NAME mgba PREFIX "lib")
endfunction()
play_add_mgba()
